rrhh$#         INSERT INTO audit_log_empleados (accion, emp_id, usuario_bd, salario_nuevo, ip_origen)

rrhh$#         RETURN NEW;
rrhh$#     ELSE
rrhh$#         INSERT INTO audit_log_empleados (accion, emp_id, usuario_bd, salario_anterior, ip_origen)
rrhh$#         VALUES ('DELETE', OLD.emp_id, session_user, OLD.salario, inet_client_addr());

rrhh$#     END IF;

rrhh=# -- 5. Mantenimiento automatico: reconstruye los indices secundarios de las particiones cuyo
rrhh=# --    nombre inicia con p_prefijo y actualiza las estadisticas de todas las particiones
rrhh=# CREATE PROCEDURE rrhh.prc_mantenimiento(p_prefijo TEXT)
rrhh-# LANGUAGE plpgsql AS
rrhh-# $$
rrhh$# DECLARE r RECORD;
rrhh$# BEGIN
rrhh$#     FOR r IN SELECT c.relname FROM pg_class c
rrhh$#              WHERE c.relnamespace = 'rrhh'::regnamespace AND c.relkind = 'i'
rrhh$#                AND c.relname !~ '_pkey$' AND c.relname !~ '_key$'
rrhh$#                AND c.relname LIKE p_prefijo || '%'
rrhh$#     LOOP
rrhh$#         EXECUTE format('REINDEX INDEX rrhh.%I', r.relname);
rrhh$#         RAISE NOTICE 'Indice reconstruido: %', r.relname;
rrhh$#     END LOOP;
rrhh$#     FOR r IN SELECT c.relname FROM pg_inherits i JOIN pg_class c ON c.oid = i.inhrelid
rrhh$#              WHERE i.inhparent = 'rrhh.empleados'::regclass
rrhh$#     LOOP
rrhh$#         EXECUTE format('ANALYZE rrhh.%I', r.relname);
rrhh$#         RAISE NOTICE 'Particion analizada: %', r.relname;
rrhh$#     END LOOP;
rrhh$# END;
rrhh$# $$;
